🗄️ DATABASE SCHEMA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

jobs table:
  - id (BIGINT PRIMARY KEY)
  - title (VARCHAR 150)
  - description (TEXT)
  - department_id (BIGINT, nullable, FK to departments)
  - position_id (BIGINT, nullable, FK to positions)
  - salary_min, salary_max (DECIMAL 12,2, nullable)
  - salary_currency (VARCHAR 3, default 'ETB')
  - employment_type (VARCHAR 50, nullable)
  - location (VARCHAR 150, nullable)
  - requirements, benefits (TEXT, nullable)
  - status (ENUM: open, closed, archived)
  - posted_date (DATE)
  - closing_date (DATE, nullable)
  - posted_by (BIGINT, nullable, FK to employees)
  - notes (TEXT, nullable)
  - timestamps, soft deletes

job_applications table:
  - id (BIGINT PRIMARY KEY)
  - job_id (BIGINT, FK to jobs ON DELETE CASCADE)
  - first_name, last_name (VARCHAR 80)
  - email (VARCHAR 100)
  - phone (VARCHAR 20)
  - resume_path (VARCHAR 255, nullable) ← FILE UPLOAD
  - cover_letter (TEXT, nullable)
  - status (ENUM: pending, shortlisted, rejected, hired)
  - applied_date (DATETIME)
  - reviewed_by (BIGINT, nullable, FK to employees)
  - reviewed_date (DATETIME, nullable)
  - notes (TEXT, nullable)
  - timestamps, soft deletes

  GET  /recruitment/jobs                    - List open jobs
  GET  /recruitment/jobs/{jobId}            - Show job details
  POST /recruitment/jobs/{jobId}/apply      - Submit application with resume

HR/SUPERADMIN ROUTES (Role-based authorization):
  GET    /recruitment/jobs                  - List all jobs
  GET    /recruitment/jobs/create           - Show create form
  POST   /recruitment/jobs                  - Create job
  GET    /recruitment/jobs/{jobId}          - Show job details
  GET    /recruitment/jobs/{jobId}/edit     - Show edit form
  PUT    /recruitment/jobs/{jobId}          - Update job
  DELETE /recruitment/jobs/{jobId}          - Delete job

  GET    /recruitment/jobs/{jobId}/applications                    - List applications
  GET    /recruitment/applications/{applicationId}                 - View application
  GET    /recruitment/applications/{applicationId}/download-resume - Download resume
  PUT    /recruitment/applications/{applicationId}/status          - Update status
  DELETE /recruitment/applications/{applicationId}                 - Delete application

📤 FILE UPLOAD DETAILS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Upload Location:
  storage/app/public/applications/{jobId}/{timestamp}_{filename}

File Naming:
  {timestamp}_{original_filename}
  Example: 1234567890_john_resume.pdf

Allowed File Types:
  - PDF (application/pdf)
  - DOC (application/msword)
  - DOCX (application/vnd.openxmlformats-officedocument.wordprocessingml.document)

File Size Limit:
  - Maximum 5MB (5120 KB)

Access Control:
  - Upload: Public (anyone can submit)
  - Download: HR/SuperAdmin only
  - Delete: HR/SuperAdmin only (automatic on application deletion)

Storage Configuration:
  - Disk: public (storage/app/public)
  - URL: /storage/applications/{jobId}/{filename}
  - Requires: php artisan storage:link
